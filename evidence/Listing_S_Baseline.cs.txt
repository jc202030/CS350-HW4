001: using System;
002: using System.Globalization;
003: 
004: public class BMICalculator
005: {
006:     public void Calculate(Person person)
007:     {
008:         double bmi;
009:         if (person.UnitSystem == "Metric")
010:         {
011:             if (person.Height < 50 || person.Height > 250)
012:             {
013:                 Console.WriteLine("Invalid measurements");
014:                 return;
015:             }
016:             if (person.Weight < 20 || person.Weight > 300)
017:             {
018:                 Console.WriteLine("Invalid measurements");
019:                 return;
020:             }
021:             bmi = CalculateMetric(person.Height, person.Weight);
022:         }
023:         else if (person.UnitSystem == "Imperial")
024:         {
025:             if (person.Height < 20 || person.Height > 100)
026:             {
027:                 Console.WriteLine("Invalid measurements");
028:                 return;
029:             }
030:             if (person.Weight < 44 || person.Weight > 660)
031:             {
032:                 Console.WriteLine("Invalid measurements");
033:                 return;
034:             }
035:             bmi = CalculateImperial(person.Height, person.Weight);
036:         }
037:         else
038:         {
039:             Console.WriteLine("Invalid unit system");
040:             return;
041:         }
042: 
043:         bmi = Math.Round(bmi, 1, MidpointRounding.AwayFromZero);
044:         string category = GetCategory(bmi);
045:         DisplayResult(person, bmi, category);
046:     }
047: 
048:     public double CalculateMetric(double height, double weight)
049:     {
050:         double meters = height / 100;
051:         return weight / (meters * meters);
052:     }
053: 
054:     public double CalculateImperial(double height, double weight)
055:     {
056:         return 703 * weight / (height * height);
057:     }
058: 
059:     public string GetCategory(double bmi)
060:     {
061:         if (bmi < 18.5)
062:             return "Underweight";
063:         else if (bmi < 25)
064:             return "Healthy";
065:         else if (bmi < 30)
066:             return "Overweight";
067:         else
068:             return "Obese";
069:     }
070: 
071:     public void DisplayResult(Person person, double bmi, string category)
072:     {
073:         Console.WriteLine("Name: " + person.Name);
074:         Console.WriteLine("BMI: " + bmi.ToString("F1", CultureInfo.InvariantCulture));
075:         Console.WriteLine("Category: " + category);
076:     }
077: }
